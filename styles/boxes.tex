% 2. 六大数学定理盒子（工厂宏统一定义，属性 100% 保持原样）
\newcommand{\NewLoneMathBox}[2]{%
	\newtcolorbox{#1}[1]{
		enhanced,
		sharp corners,
		colback=qianbai,
		colframe=#2,
		boxrule=2.5pt,
		fontupper={\kaishu},
		halign=justify,
		breakable,
		title={\TitleBoxII{#2_1}{##1}{#2_2}{heise}{mibai}},
		attach boxed title to top left={yshift=-2.5pt, xshift=0mm},
		boxed title style={
			boxrule=0pt,
			boxsep=0pt,
			top=-2.1pt,
			bottom=0pt,
			left=0pt,
			right=0pt,
			colback=qianbai,
			baseline=\ht\strutbox
		},
		after upper={\hfill\includegraphics[height=8pt]{assets/pdf/mark_#2.pdf}},
	}%
}
\NewLoneMathBox{definition}{definition}
\NewLoneMathBox{theorem}{theorem}
\NewLoneMathBox{proposition}{proposition}
\NewLoneMathBox{corollary}{corollary}
\NewLoneMathBox{lemma}{lemma}
\NewLoneMathBox{example}{example}

		\node[inner sep=0pt, outer sep=0pt] (char) {\phantom{#1}};%
		\fill[qiancheng] (char.south west) rectangle ([xshift=0pt, yshift=1.5pt]char.south east);%
		\node[inner sep=0pt, outer sep=0pt] (char) {#1};%
	}%
}
